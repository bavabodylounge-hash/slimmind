⚙️ scoreAxes() — 핵심 채점 함수 survey-hospital.html L16873~L16975
function scoreAxes(){
  // ① 축별 점수 초기화
  const sc = {A01:0, A02:0, ..., A10:0};
  const got = {}, max = {};

  // ② 각 문항별 원점수 수집
  for(let gi=0; gi < AXIS_ORDER.length; gi++){
    const ax = AXIS_ORDER[gi];          // 해당 문항의 축 코드
    const v  = stage3Answers[gi];       // 사용자 선택 인덱스
    const q  = stage3Flat[gi];

    // 기질문항(scores=null) → 채점 완전 제외
    if(q.scores === null) continue;

    // 선택지형(q.scores = 배열) → 직접 매핑
    if(Array.isArray(q.scores)){
      got[ax] += q.scores[v];
      max[ax] += Math.max(...q.scores);
      continue;
    }

    // 일반문항 (0~3 리커트)
    const raw = (typeof v==='number') ? v : 0;
    const val = (q.rev) ? (3 - raw) : raw;   // 역문항: 3-v 반전
    got[ax] += val;
    max[ax] += 3;
  }

  // ③ 0~10 스케일 정규화
  Object.keys(sc).forEach(ax => {
    if(!max[ax]){ sc[ax]=0; return; }
    sc[ax] = Math.round(got[ax] / max[ax] * 10 * 100) / 100;
  });

  // ④ A10 기질 반감 (×0.5)
  sc.A10 = Math.round(sc.A10 * 0.5 * 100) / 100;

  // ⑤ 병력 보너스 가산 (캡 +4.0)
  try {
    const s2 = stage2Bonus();
    Object.keys(s2).forEach(ax => { if(sc[ax]!==undefined) sc[ax] += s2[ax]; });
  } catch(e) {}

  // ⑥ 프로필 보너스 가산 (캡 +2.5)
  try {
    const pf = pfBonus();
    Object.keys(pf).forEach(ax => { if(sc[ax]!==undefined) sc[ax] += pf[ax]; });
  } catch(e) {}

  // ⑦ 최종 값 저장 → applyDecision() 호출
  return sc; // { A01:7.5, A02:3.2, ..., A10:2.1 }
}

0~10 스케일sc[ax] = round(got[ax] / max[ax] × 10, 2)0.00~10.00
A10 반감sc.A10 = sc.A10 × 0.50.00~5.00
병력보너스(s2)sc[ax] += s2[ax] (캡 +4.0)최대 +4.0
프로필보너스(pf)sc[ax] += pf[ax] (캡 +2.5)최대 +2.5
최종 상한약 10 + 4 + 2.5 = 16.5이론적 최대
